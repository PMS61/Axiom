% ---- Algorithm boxes: \begin{algobox}{Title} ... \end{algobox} ----
\usepackage[most]{tcolorbox}

\renewcommand{\thealgocounter}{\thechapter.\arabic{algocounter}}
\newenvironment{algobox}[1]{%
  \refstepcounter{algocounter}%
  \begin{tcolorbox}[
    enhanced, breakable,
    colback=white, colframe=darkblue!70!black,
    coltitle=white, fonttitle=\bfseries,
    title={Algorithm \thealgocounter: #1},
    boxrule=0.6pt, arc=2pt,
    left=6pt, right=6pt, top=4pt, bottom=4pt]%
}{%
  \end{tcolorbox}%
}

% ---- Headings ----
% Numbered chapter: one centred bold line "Chapter N : Title" (\@chapapp is "Appendix" after \appendix,
% which gives "Appendix A : Title"). Starred chapters (Contents, lists, Bibliography) are centred, unnumbered.
\makeatletter
\titleformat{\chapter}[block]
  {\normalfont\bfseries\fontsize{16}{20}\selectfont\centering}
  {\@chapapp\ \thechapter\ :}{0.5em}
  {\fontsize{16}{20}\selectfont}
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\bfseries\fontsize{16}{20}\selectfont\centering}
  {}{0pt}
  {\fontsize{16}{20}\selectfont}
\titlespacing*{\chapter}{0pt}{6pt}{20pt}
\titleformat{\section}
  {\normalfont\bfseries\fontsize{14}{17}\selectfont\raggedright}
  {\thesection}{0.6em}{}
\titlespacing*{\section}{0pt}{18pt}{10pt}
\titleformat{\subsection}
  {\normalfont\bfseries\fontsize{12}{15}\selectfont\raggedright}
  {\thesubsection}{0.6em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}
\makeatother
