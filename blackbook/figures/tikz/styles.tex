% Stick-figure actor centred at #1 (a coordinate), label #2 placed below it.
\newcommand{\StickActor}[2]{%
  \begin{scope}[shift={#1}]
    \draw[actorline] (0,0.62) circle (0.13);
    \draw[actorline] (0,0.49) -- (0,0.08);
    \draw[actorline] (-0.22,0.38) -- (0.22,0.38);
    \draw[actorline] (0,0.08) -- (-0.18,-0.26);
    \draw[actorline] (0,0.08) -- (0.18,-0.26);
    \node[actor, anchor=north] at (0,-0.32) {#2};
  \end{scope}%
}
